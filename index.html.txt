      socket.onerror=()=>{if(isCurrent()){setStatus('Feed connection error');if($('scanDetail'))$('scanDetail').textContent='Could not reach Deriv live data. Check internet access or try again shortly.';}};
      socket.onclose=()=>{
        if(!isCurrent())return;
        if(pingTimer){clearInterval(pingTimer);pingTimer=null;}
        $('predictBtn').disabled=true;
        if(!manuallyStopped){setStatus('Disconnected — retrying');scheduleReconnect();}
        else setStatus('Feed stopped');
      };
    }catch(e){setStatus('Feed unavailable');if($('scanDetail'))$('scanDetail').textContent='Browser could not open the market feed: '+(e.message||'connection unavailable');scheduleReconnect();}
  }
  function digitFromQuote(quote,pipSize){
    const n=Number(quote);if(!Number.isFinite(n))return null;
    const decimals=Math.max(0,Math.min(Number.isInteger(pipSize)?pipSize:2,8));
    const formatted=n.toFixed(decimals);
    const match=formatted.match(/(\d)$/);
    return match?Number(match[1]):null;
  }
  function scheduleReconnect(){
    if(manuallyStopped || reconnectTimer || !navigator.onLine)return;
    reconnectAttempts++;
    const delay=Math.min(15000,1000*Math.pow(2,Math.min(reconnectAttempts,4)));
    reconnectTimer=setTimeout(()=>{reconnectTimer=null;connectFeed();},delay);
  }
  function stopFeed(){
    manuallyStopped=true;feedGeneration++;
    if(pingTimer){clearInterval(pingTimer);pingTimer=null;}
    if(reconnectTimer){clearTimeout(reconnectTimer);reconnectTimer=null;}
    if(ws){try{ws.onclose=null;ws.close()}catch(e){}ws=null;}
    if(predictionTimer)clearTimeout(predictionTimer);
    if(countdownInterval)clearInterval(countdownInterval);
    $('predictBtn').disabled=true;
  }

  function renderAnalysis(){
    const c=counts(), n=ticks.length, grid=$('frequencyGrid');grid.innerHTML='';
    const max=Math.max(1,...c);
    c.forEach((v,d)=>{const el=document.createElement('div');el.className='digitbox';if(v===max&&n>0)el.classList.add('hot');el.innerHTML='<b>'+d+'</b><small>'+ (n?((v/n)*100).toFixed(1):'0.0')+'%</small>';grid.appendChild(el);});
    $('sampleCount').textContent=n+' ticks';
    $('history').innerHTML=ticks.slice(0,35).map(d=>'<span class="tick">'+d+'</span>').join('')||'<span class="muted small">Waiting for live ticks…</span>';
    if(n){const sorted=c.map((v,d)=>({d,v})).sort((a,b)=>b.v-a.v);$('mostFrequent').textContent=sorted[0].d+' ('+sorted[0].v+' ticks)';$('leastFrequent').textContent=sorted[sorted.length-1].d+' ('+sorted[sorted.length-1].v+' ticks)';}
    $('predictBtn').disabled=n<20 || !ws || ws.readyState!==WebSocket.OPEN || scanInProgress;
    if(!scanInProgress && $('scanState') && n<20) setScanProgress((n/20)*100,'Collecting data',`Deep scan needs at least 20 live ticks; currently ${n}/20.`);
  }
  function setScanProgress(progress, state, detail){
    const p=Math.max(0,Math.min(100,progress));
    if($('scanBar')) $('scanBar').style.width=p+'%';
    if($('scanState')) $('scanState').textContent=state;
    if($('scanDetail')) $('scanDetail').textContent=detail;
  }
  function evaluateMarket(){
    const n=ticks.length;
    if(n<20){
      lastScan=null;
      setScanProgress(Math.min(100,(n/20)*100),'Collecting data',`Deep scan needs at least 20 live ticks; currently ${n}/20.`);
      $('predictBtn').disabled=true;
      return null;
    }
    const all=counts();
    const recent=ticks.slice(0,20), older=ticks.slice(20,Math.min(n,150));
    const recentCounts=Array(10).fill(0), olderCounts=Array(10).fill(0);
    recent.forEach(d=>recentCounts[d]++);
    older.forEach(d=>olderCounts[d]++);
    const total=recent.length, oldTotal=older.length;
    const recentRates=recentCounts.map(v=>v/total);
    const expected=1/10;
    const deviations=recentRates.map((r,d)=>({d,rate:r,count:recentCounts[d],deviation:Math.abs(r-expected),olderRate:oldTotal?olderCounts[d]/oldTotal:expected}));
    // Ranking is descriptive only: recent sample frequency and consistency, not a claim that a digit is due.
    deviations.sort((a,b)=>b.count-a.count || b.deviation-a.deviation);
    const best=deviations[0];
    const stability=oldTotal?Math.max(0,1-Math.abs(best.rate-best.olderRate)):0;
    const sampleQuality=Math.min(1,n/150);
    const descriptiveScore=Math.round(Math.min(99,Math.max(1,(best.rate*100)*0.65+(stability*100)*0.2+sampleQuality*15)));
    lastScan={digit:best.d,frequency:best.rate*100,score:descriptiveScore,n,at:Date.now()};
    setScanProgress(100,'Scan complete',`Reviewed ${n} ticks across recent and earlier samples. Top historical digit: ${best.d}; recent frequency ${(best.rate*100).toFixed(1)}%. The percentage shown is historical sample frequency, not a verified win probability or guarantee.`);
    return lastScan;
  }
  function runDeepScan(){
    if(scanInProgress)return;
    scanInProgress=true; scanProgress=0;
    $('predictBtn').disabled=true;
    $('predDigit').textContent='—';
    
    setScanProgress(0,'Scanning…','Reviewing live tick sample, digit distribution, and recent-vs-older frequency.');
    const started=Date.now();
    if(scanInterval)clearInterval(scanInterval);
    scanInterval=setInterval(()=>{
      scanProgress=Math.min(95,((Date.now()-started)/1800)*95);
      setScanProgress(scanProgress,'Scanning…','Reviewing live tick sample, digit distribution, and recent-vs-older frequency.');
    },100);
    setTimeout(()=>{
      if(scanInterval)clearInterval(scanInterval);
      let result=null;
      try { result=evaluateMarket(); }
      catch(err) { setScanProgress(0,'Scan error','The scan could not finish. Please wait for live ticks and try again.'); console.error('ZUITCHINI scan error:',err); }
      scanInProgress=false;
      if(result){
        $('predDigit').textContent=result.digit;
        if ($('scanDetail')) $('scanDetail').textContent='Scan complete for '+selectedName+'. Candidate digit is based on historical frequency, not a verified win probability.';
        $('digitPercent').textContent=result.frequency.toFixed(1)+'%';
        predictionExpires=Date.now()+30000;
        if(predictionTimer)clearTimeout(predictionTimer);
        if(countdownInterval)clearInterval(countdownInterval);
        const update=()=>{const left=Math.max(0,predictionExpires-Date.now());$('countdown').textContent=(left/1000).toFixed(1)+'s';if(left<=0){clearInterval(countdownInterval);$('digitPercent').textContent='—%';$('countdown').textContent='Expired';}};
        update();countdownInterval=setInterval(update,100);
        predictionTimer=setTimeout(()=>{$('predDigit').textContent='—';$('digitPercent').textContent='—%';},30000);
      }else{
        $('predDigit').textContent='—';
        $('digitPercent').textContent='—%';
      }
      renderAnalysis();
    },1900);
  }

  $('predictBtn').addEventListener('click',runDeepScan);

  // Feed starts after login; the browser retries after internet returns.
  setStatus('Ready to sign in');
